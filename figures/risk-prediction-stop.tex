  \node[offline] (train) at (0,26mm) {\textbf{离线训练风险判别模型}\\\(\pi_{0.5}\)冻结，只优化\(\boldsymbol\psi\)};
  \node[offline] (thr) at (-46mm,-26mm) {\textbf{停止判据研究}\\验证集比较\\预警及时性与误报};
